        echo "<tr>
            <td data-label='ID'>{$row['id']}</td>
            <td data-label='Name'>{$row['crop_name']}</td>
            <td data-label='Price'>{$row['price']}</td>
            <td data-label='Quantity'>{$row['quantity']}</td>
            <td data-label='Buy'>
                <form method='post' class='d-flex justify-content-center align-items-center gap-2'>
                    <input type='hidden' name='id' value='{$row['id']}'>
                    <input type='number' name='buy_qty' placeholder='kg' required>
                    <button type='submit' name='buy' class='btn-buy'>Buy</button>
                </form>
            </td>
        </tr>";
    }

    echo "</tbody></table></div>";

    if (isset($_POST['buy'])) {
        $id = $_POST['id'];
        $buy_qty = $_POST['buy_qty'];

        // Fetch current quantity
        $res = $conn->query("SELECT quantity FROM crops WHERE id = $id");
        $data = $res->fetch_assoc();

        if ($buy_qty <= $data['quantity']) {
            $new_qty = $data['quantity'] - $buy_qty;
            $conn->query("UPDATE crops SET quantity = $new_qty WHERE id = $id");
            echo "<div class='message text-success'>✅ Purchase successful!</div>";
        } else {
            echo "<div class='message text-danger'>❌ Not enough stock!</div>";
        }
