package com.example.inspiretools

import android.Manifest
import android.annotation.SuppressLint
import android.app.Activity
import android.content.Context
import android.content.Intent
import android.content.SharedPreferences
import android.content.pm.PackageManager
import android.graphics.Bitmap
import android.graphics.BitmapFactory
import android.net.Uri
import android.os.Bundle
import android.provider.MediaStore
import android.view.MenuItem
import android.widget.*
import androidx.appcompat.app.AlertDialog
import androidx.appcompat.app.AppCompatActivity
import androidx.appcompat.widget.Toolbar
import androidx.core.app.ActivityCompat
import androidx.core.content.ContextCompat

import java.io.File
import java.io.FileOutputStream
import java.io.InputStream
import java.text.SimpleDateFormat
import java.util.*

class AdminInfoActivity : AppCompatActivity() {

    private lateinit var profileImage: ImageView
    private lateinit var adminNameText: TextView
    private lateinit var adminEmailText: TextView
    private lateinit var adminPhoneText: TextView
    private lateinit var adminDesignationText: TextView
    private lateinit var adminDepartmentText: TextView
    private lateinit var adminJoinDateText: TextView
    private lateinit var adminLocationText: TextView
    private lateinit var adminBioText: TextView

    private lateinit var editProfileButton: Button
    private lateinit var changePhotoButton: Button
    private lateinit var adminStatsCard: LinearLayout
    private lateinit var totalUsersText: TextView
    private lateinit var activeSessionsText: TextView
    private lateinit var lastLoginText: TextView

    private lateinit var sharedPreferences: SharedPreferences
    private var profileImagePath = ""

    companion object {
        private const val IMAGE_PICK_REQUEST = 2001
        private const val CAMERA_REQUEST = 2002
        private const val STORAGE_PERMISSION_REQUEST = 2003
        private const val CAMERA_PERMISSION_REQUEST = 2004
    }

    @SuppressLint("MissingInflatedId")
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_admin_info)
        findViewById<ImageView>(R.id.backButton).setOnClickListener {
            finish() // goes back to previous Activity
        }

        sharedPreferences = getSharedPreferences("admin_profile", Context.MODE_PRIVATE)

        setupToolbar()
        initializeViews()
        loadAdminData()
        setupClickListeners()
    }

    private fun setupToolbar() {
        val toolbar = findViewById<Toolbar>(R.id.toolbar)
        setSupportActionBar(toolbar)

        supportActionBar?.apply {
            title = "Admin Profile"
            setDisplayHomeAsUpEnabled(true)
            setDisplayShowHomeEnabled(true)
        }
    }

    private fun initializeViews() {
        profileImage = findViewById(R.id.profileImage)
        adminNameText = findViewById(R.id.adminNameText)
        adminEmailText = findViewById(R.id.adminEmailText)
        adminPhoneText = findViewById(R.id.adminPhoneText)
        adminDesignationText = findViewById(R.id.adminDesignationText)
        adminDepartmentText = findViewById(R.id.adminDepartmentText)
        adminJoinDateText = findViewById(R.id.adminJoinDateText)
        adminLocationText = findViewById(R.id.adminLocationText)
        adminBioText = findViewById(R.id.adminBioText)

        editProfileButton = findViewById(R.id.editProfileButton)
        changePhotoButton = findViewById(R.id.changePhotoButton)
        adminStatsCard = findViewById(R.id.adminStatsCard)
        totalUsersText = findViewById(R.id.totalUsersText)
        activeSessionsText = findViewById(R.id.activeSessionsText)
        lastLoginText = findViewById(R.id.lastLoginText)
    }

    private fun loadAdminData() {
        adminNameText.text = sharedPreferences.getString("admin_name", "John Smith")
        adminEmailText.text = sharedPreferences.getString("admin_email", "admin@inspiretools.com")
        adminPhoneText.text = sharedPreferences.getString("admin_phone", "+91 98765 43210")
        adminDesignationText.text = sharedPreferences.getString("admin_designation", "System Administrator")
        adminDepartmentText.text = sharedPreferences.getString("admin_department", "IT Department")
        adminJoinDateText.text = sharedPreferences.getString("admin_join_date", "01 Jan 2020")
        adminLocationText.text = sharedPreferences.getString("admin_location", "Mumbai, Maharashtra")
        adminBioText.text = sharedPreferences.getString("admin_bio",
            "Inspire Tool is an advanced productivity app powered by ML Kit. It provides multiple smart tools such as Translator Helper, Image Resizer, Math Solver, Text Scanner, and Object Detector. Our aim is to make everyday tasks simpler with AI-powered solutions.")

        profileImagePath = sharedPreferences.getString("admin_profile_image", "") ?: ""
        if (profileImagePath.isNotEmpty()) {
            loadProfileImage(profileImagePath)
        }

        loadAdminStats()
    }

    private fun loadAdminStats() {
        val totalUsers = sharedPreferences.getInt("total_users", 156)
        val activeSessions = sharedPreferences.getInt("active_sessions", 23)
        val lastLoginTime = sharedPreferences.getString("last_login", getCurrentDateTime())

        totalUsersText.text = totalUsers.toString()
        activeSessionsText.text = activeSessions.toString()
        lastLoginText.text = lastLoginTime
    }

    private fun setupClickListeners() {
        editProfileButton.setOnClickListener {
            showEditProfileDialog()
        }

        changePhotoButton.setOnClickListener {
            showImagePickerDialog()
        }

        profileImage.setOnClickListener {
            showImagePickerDialog()
        }

        findViewById<LinearLayout>(R.id.contactInfoCard).setOnClickListener {
            showContactOptionsDialog()
        }

        findViewById<LinearLayout>(R.id.workInfoCard).setOnClickListener {
            showWorkInfoDialog()
        }
    }

    private fun showEditProfileDialog() {
        val dialogView = LinearLayout(this).apply {
            orientation = LinearLayout.VERTICAL
            setPadding(48, 32, 48, 32)
        }

        val nameInput = EditText(this).apply {
            hint = "Name"
            setText(adminNameText.text.toString())
        }
        val emailInput = EditText(this).apply {
            hint = "Email"
            setText(adminEmailText.text.toString())
        }
        val phoneInput = EditText(this).apply {
            hint = "Phone"
            setText(adminPhoneText.text.toString())
        }
        val designationInput = EditText(this).apply {
            hint = "Designation"
            setText(adminDesignationText.text.toString())
        }
        val departmentInput = EditText(this).apply {
            hint = "Department"
            setText(adminDepartmentText.text.toString())
        }
        val locationInput = EditText(this).apply {
            hint = "Location"
            setText(adminLocationText.text.toString())
        }
        val bioInput = EditText(this).apply {
            hint = "Bio"
            setText(adminBioText.text.toString())
            minLines = 3
        }

        dialogView.addView(nameInput)
        dialogView.addView(emailInput)
        dialogView.addView(phoneInput)
        dialogView.addView(designationInput)
        dialogView.addView(departmentInput)
        dialogView.addView(locationInput)
        dialogView.addView(bioInput)

        AlertDialog.Builder(this)
            .setTitle("Edit Profile")
            .setView(dialogView)
            .setPositiveButton("Save") { _, _ ->
                saveAdminData(
                    nameInput.text.toString(),
                    emailInput.text.toString(),
                    phoneInput.text.toString(),
                    designationInput.text.toString(),
                    departmentInput.text.toString(),
                    locationInput.text.toString(),
                    bioInput.text.toString()
                )
                loadAdminData()
                showToast("Profile updated successfully")
            }
            .setNegativeButton("Cancel", null)
            .show()
    }

    private fun saveAdminData(name: String, email: String, phone: String,
                              designation: String, department: String,
                              location: String, bio: String) {
        with(sharedPreferences.edit()) {
            putString("admin_name", name)
            putString("admin_email", email)
            putString("admin_phone", phone)
            putString("admin_designation", designation)
            putString("admin_department", department)
            putString("admin_location", location)
            putString("admin_bio", bio)
            apply()
        }
    }

    private fun showImagePickerDialog() {
        val options = arrayOf("Take Photo", "Choose from Gallery", "Remove Photo")

        AlertDialog.Builder(this)
            .setTitle("Change Profile Photo")
            .setItems(options) { _, which ->
                when (which) {
                    0 -> checkCameraPermissionAndCapture()
                    1 -> checkStoragePermissionAndPick()
                    2 -> removeProfilePhoto()
                }
            }
            .show()
    }

    private fun checkCameraPermissionAndCapture() {
        if (ContextCompat.checkSelfPermission(this, Manifest.permission.CAMERA)
            == PackageManager.PERMISSION_GRANTED) {
            openCamera()
        } else {
            ActivityCompat.requestPermissions(
                this,
                arrayOf(Manifest.permission.CAMERA),
                CAMERA_PERMISSION_REQUEST
            )
        }
    }

    private fun checkStoragePermissionAndPick() {
        if (android.os.Build.VERSION.SDK_INT >= android.os.Build.VERSION_CODES.TIRAMISU) {
            openImagePicker()
        } else {
            if (ContextCompat.checkSelfPermission(this, Manifest.permission.READ_EXTERNAL_STORAGE)
                == PackageManager.PERMISSION_GRANTED) {
                openImagePicker()
            } else {
                ActivityCompat.requestPermissions(
                    this,
                    arrayOf(Manifest.permission.READ_EXTERNAL_STORAGE),
                    STORAGE_PERMISSION_REQUEST
                )
            }
        }
    }

    private fun openCamera() {
        val intent = Intent(MediaStore.ACTION_IMAGE_CAPTURE)
        if (intent.resolveActivity(packageManager) != null) {
            startActivityForResult(intent, CAMERA_REQUEST)
        } else {
            showToast("Camera not available")
        }
    }

    private fun openImagePicker() {
        val intent = Intent(Intent.ACTION_PICK).apply {
            type = "image/*"
        }
        if (intent.resolveActivity(packageManager) != null) {
            startActivityForResult(Intent.createChooser(intent, "Select Profile Photo"), IMAGE_PICK_REQUEST)
        } else {
            showToast("Gallery not available")
        }
    }

    private fun removeProfilePhoto() {
        profileImage.setImageResource(R.drawable.ic_admin_default)
        profileImagePath = ""
        with(sharedPreferences.edit()) {
            putString("admin_profile_image", "")
            apply()
        }
        showToast("Profile photo removed")
    }

    private fun loadProfileImage(imagePath: String) {
        try {
            val file = File(imagePath)
            if (file.exists()) {
                val bitmap = BitmapFactory.decodeFile(imagePath)
                profileImage.setImageBitmap(bitmap)
            }
        } catch (e: Exception) {
            profileImage.setImageResource(R.drawable.ic_admin_default)
        }
    }

    private fun saveImageToInternalStorage(bitmap: Bitmap): String {
        return try {
            val filename = "admin_profile_${System.currentTimeMillis()}.jpg"
            val file = File(filesDir, filename)
            val fos = FileOutputStream(file)
            bitmap.compress(Bitmap.CompressFormat.JPEG, 90, fos)
            fos.close()
            file.absolutePath
        } catch (e: Exception) {
            ""
        }
    }

    private fun showContactOptionsDialog() {
        val options = arrayOf("Call", "Email", "Copy Phone", "Copy Email")

        AlertDialog.Builder(this)
            .setTitle("Contact")
            .setItems(options) { _, which ->
                when (which) {
                    0 -> {
                        val intent = Intent(Intent.ACTION_DIAL, Uri.parse("tel:${adminPhoneText.text}"))
                        startActivity(intent)
                    }
                    1 -> {
                        val intent = Intent(Intent.ACTION_SENDTO, Uri.parse("mailto:${adminEmailText.text}"))
                        startActivity(intent)
                    }
                    2 -> {
                        copyToClipboard("Phone", adminPhoneText.text.toString())
                        showToast("Phone copied")
                    }
                    3 -> {
                        copyToClipboard("Email", adminEmailText.text.toString())
                        showToast("Email copied")
                    }
                }
            }
            .show()
    }

    private fun showWorkInfoDialog() {
        val workInfo = """
            Department: ${adminDepartmentText.text}
            Designation: ${adminDesignationText.text}
            Join Date: ${adminJoinDateText.text}
            Location: ${adminLocationText.text}
            
            Responsibilities:
            • User Management & Access Control
            • System Configuration & Updates
            • Data Security & Backup Management
            • Technical Support & Troubleshooting
            • Performance Monitoring & Analytics
        """.trimIndent()

        AlertDialog.Builder(this)
            .setTitle("Work Information")
            .setMessage(workInfo)
            .setPositiveButton("OK", null)
            .show()
    }

    private fun copyToClipboard(label: String, text: String) {
        val clipboard = getSystemService(Context.CLIPBOARD_SERVICE) as android.content.ClipboardManager
        val clip = android.content.ClipData.newPlainText(label, text)
        clipboard.setPrimaryClip(clip)
    }

    private fun getCurrentDateTime(): String {
        return SimpleDateFormat("dd MMM yyyy, HH:mm", Locale.getDefault()).format(Date())
    }

    private fun showToast(message: String) {
        Toast.makeText(this, message, Toast.LENGTH_SHORT).show()
    }

    override fun onRequestPermissionsResult(
        requestCode: Int,
        permissions: Array<out String>,
        grantResults: IntArray
    ) {
        super.onRequestPermissionsResult(requestCode, permissions, grantResults)

        when (requestCode) {
            STORAGE_PERMISSION_REQUEST -> {
                if (grantResults.isNotEmpty() && grantResults[0] == PackageManager.PERMISSION_GRANTED) {
                    openImagePicker()
                } else {
                    showToast("Permission required to access storage")
                }
            }
            CAMERA_PERMISSION_REQUEST -> {
                if (grantResults.isNotEmpty() && grantResults[0] == PackageManager.PERMISSION_GRANTED) {
                    openCamera()
                } else {
                    showToast("Permission required to access camera")
                }
            }
        }
    }

    @Deprecated("Deprecated in Java")
    override fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
        super.onActivityResult(requestCode, resultCode, data)

        if (resultCode == Activity.RESULT_OK) {
            when (requestCode) {
                IMAGE_PICK_REQUEST -> {
                    data?.data?.let { uri ->
                        try {
                            val inputStream: InputStream? = contentResolver.openInputStream(uri)
                            val bitmap = BitmapFactory.decodeStream(inputStream)
                            inputStream?.close()

                            if (bitmap != null) {
                                profileImage.setImageBitmap(bitmap)
                                profileImagePath = saveImageToInternalStorage(bitmap)
                                with(sharedPreferences.edit()) {
                                    putString("admin_profile_image", profileImagePath)
                                    apply()
                                }
                                showToast("Profile photo updated")
                            } else {
                                showToast("Failed to load image")
                            }
                        } catch (e: Exception) {
                            showToast("Error loading image")
                        }
                    }
                }
                CAMERA_REQUEST -> {
                    data?.extras?.get("data")?.let { bitmap ->
                        if (bitmap is Bitmap) {
                            profileImage.setImageBitmap(bitmap)
                            profileImagePath = saveImageToInternalStorage(bitmap)
                            with(sharedPreferences.edit()) {
                                putString("admin_profile_image", profileImagePath)
                                apply()
                            }
                            showToast("Profile photo updated")
                        }
                    }
                }
            }
        }
    }

    override fun onOptionsItemSelected(item: MenuItem): Boolean {
        return when (item.itemId) {
            android.R.id.home -> {
                finish()
                true
            }
            else -> super.onOptionsItemSelected(item)
        }
    }
}
